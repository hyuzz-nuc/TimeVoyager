## 三、Electron 桌面打包

### 3.1 前置准备

#### 创建 Electron 主进程文件
位置：`electron/main.js`

```javascript
const { app, BrowserWindow } = require('electron')
const path = require('path')

let mainWindow

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1200,
    height: 800,
    minWidth: 800,
    minHeight: 600,
    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,
      preload: path.join(__dirname, 'preload.js')
    },
    icon: path.join(__dirname, '../public/icons.svg'),
    frame: true,
    backgroundColor: '#0f172a'
  })

  // 开发模式
  if (process.env.NODE_ENV === 'development') {
    mainWindow.loadURL('http://localhost:5173')
    mainWindow.webContents.openDevTools()
  } else {
    // 生产模式
    mainWindow.loadFile(path.join(__dirname, '../dist/index.html'))
  }

  mainWindow.on('closed', () => {
    mainWindow = null
  })
}

app.whenReady().then(createWindow)

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') {
    app.quit()
  }
})

app.on('activate', () => {
  if (BrowserWindow.getAllWindows().length === 0) {
    createWindow()
  }
})
```

#### 创建预加载脚本
位置：`electron/preload.js`

```javascript
const { contextBridge, ipcRenderer } = require('electron')

contextBridge.exposeInMainWorld('electronAPI', {
  platform: process.platform,
  versions: {
    node: process.versions.node,
    chrome: process.versions.chrome,
    electron: process.versions.electron
  }
})
```

### 3.2 配置 electron-builder

在 `package.json` 中添加：

```json
"build": {
  "appId": "com.timevoyager.app",
  "productName": "TimeVoyager — 时光旅行者",
  "directories": {
    "output": "release",
    "buildResources": "public"
  },
  "files": [
    "dist/**/*",
    "electron/**/*"
  ],
  "extraMetadata": {
    "main": "electron/main.js"
  },
  "win": {
    "target": [
      {
        "target": "nsis",
        "arch": ["x64"]
      }
    ],
    "icon": "public/icons.svg"
  },
  "nsis": {
    "oneClick": false,
    "allowToChangeInstallationDirectory": true,
    "createDesktopShortcut": true,
    "createStartMenuShortcut": true,
    "shortcutName": "TimeVoyager"
  },
  "mac": {
    "target": [
      {
        "target": "dmg",
        "arch": ["x64", "arm64"]
      }
    ],
    "icon": "public/icons.svg",
    "category": "public.app-category.productivity"
  },
  "linux": {
    "target": ["AppImage"],
    "icon": "public/icons.svg"
  }
}
```

### 3.3 打包命令

#### 仅打包当前系统
```bash
npm run electron:build
```

#### 打包 Windows
```bash
npx electron-builder --win
```

#### 打包 macOS
```bash
npx electron-builder --mac
```

#### 打包 Linux
```bash
npx electron-builder --linux
```

#### 打包所有平台
```bash
npx electron-builder --win --mac --linux
```

## 六、代码签名（可选）

### 6.1 Windows 代码签名
需要购买代码签名证书（如 DigiCert、GlobalSign）

在 `package.json` 中配置：
```json
"build": {
  "win": {
    "certificateSubjectName": "Your Company Name",
    "rfc3161TimeStampServer": "http://timestamp.digicert.com"
  }
}
```

### 6.2 macOS 代码签名
需要 Apple Developer 账号

```bash
# 设置签名身份
export CSC_NAME="Your Developer ID"

# 打包
npm run electron:build
```

## 九、常见问题

### 9.1 Electron 打包体积过大
**原因：** 包含完整 Chromium  
**解决：** 接受现实，Electron 应用 100MB+ 是正常的

### 9.2 Windows 安装被拦截
**原因：** 未代码签名  
**解决：** 购买证书签名，或告知用户点击"更多信息"→"仍要运行"

### 9.3 Mac 无法打开
**原因：** 未签名/公证  
**解决：** 
```bash
xattr -cr /Applications/TimeVoyager.app
```
或在系统偏好设置→安全性中允许

### 9.4 PWA 无法安装
**原因：** 
- 不是 HTTPS
- 缺少 icons
- manifest 配置错误

**解决：**
- 使用 HTTPS 部署
- 生成正确尺寸的图标
- 检查 manifest 配置
